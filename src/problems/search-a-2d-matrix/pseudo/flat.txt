function searchMatrix(matrix, target)
  rows = number of rows
  cols = number of columns
  l = 0                          # [1] treat the grid as one sorted line
  r = rows * cols - 1            # [1] of rows * cols items
  while l <= r                   # [2] something is still live
    m = l + (r - l) / 2          # [2] middle of the flat line
    row = m / cols               # [2] flat index -> row (rounded down)
    col = m mod cols             # [2] flat index -> column
    if matrix[row][col] == target
      return true                # [4] found it: that is the answer
    if matrix[row][col] < target
      l = m + 1                  # [3] too small: drop the left part
    else
      r = m - 1                  # [3] too big: drop the right part
  return false                   # [4] live range is empty: not found
