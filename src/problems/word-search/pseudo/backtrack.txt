function exist(board, word)
  function dfs(r, c, k)
    if k == length of word                   # [2] every letter is matched
      return true
    if (r, c) is off the board or board[r][c] != word[k]
      return false                           # [2] off the board, in use, or wrong letter
    saved = board[r][c]
    board[r][c] = "#"                        # [3] mark the cell as in use
    found = dfs(r, c+1, k+1) or dfs(r+1, c, k+1)
         or dfs(r, c-1, k+1) or dfs(r-1, c, k+1)   # [3] try the four neighbours
    board[r][c] = saved                      # [4] restore the letter
    return found

  for each cell (r, c)
    if dfs(r, c, 0)                          # [1] a start cell worked
      return true
  return false                               # [1] no start cell worked
