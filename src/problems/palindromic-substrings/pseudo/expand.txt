function countSubstrings(s)
  count = 0                             # [1] palindromes found so far
  for c from 0 to length(s) - 1
    for w in [0, 1]                     # 0 = odd centre on c, 1 = even centre between c and c+1
      l = c, r = c + w
      while l >= 0 and r < length(s) and s[l] == s[r]   # [2] ends match: one more palindrome
        count = count + 1
        l = l - 1                       # [2] grow outward
        r = r + 1
      # [3] growth stopped: this centre is finished
  return count                          # [4] total over all centres
