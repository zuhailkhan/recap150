class KthLargest(k, nums)
  heap = empty min-heap                     # smallest kept number on top
  for x in nums: add(x)                     # [1] start with the numbers we were given

  add(val)
    push val into heap                      # [2] every new number goes in first
    if size(heap) > k
      pop the smallest from heap            # [3] one too many: it cannot be the kth largest
    return top of heap                      # [4] the top is the answer
