function plusOne(digits)
  out = copy of digits
  for i from last index down to 0       # [1] start at the last digit and walk left
    if out[i] < 9
      out[i] = out[i] + 1               # [3] a digit below 9 absorbs the carry, so we are done
      return out
    out[i] = 0                          # [2] 9 + 1 = 10: write 0 and carry 1 to the left
  return [1] followed by out            # [3] all nines: the result is 1 followed by zeros
