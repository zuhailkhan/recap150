function minDistance(word1, word2)
  m = length of word1, n = length of word2
  dp = (m+1) x (n+1) table of 0s            # dp[i][j] = edits: first i letters of word1 -> first j of word2
  for i from 0 to m: dp[i][0] = i           # [1] delete i letters to reach the empty word
  for j from 0 to n: dp[0][j] = j           # [1] insert j letters from the empty word
  for i from 1 to m
    for j from 1 to n
      if word1[i-1] == word2[j-1]
        dp[i][j] = dp[i-1][j-1]             # [2] same letter: no edit, copy the diagonal
      else
        # [3] one edit plus the cheapest of: delete (up), insert (left), replace (diagonal)
        dp[i][j] = 1 + min(dp[i-1][j], dp[i][j-1], dp[i-1][j-1])
  return dp[m][n]                           # [4] whole word1 to whole word2
