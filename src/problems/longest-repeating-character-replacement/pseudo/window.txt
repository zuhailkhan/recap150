function characterReplacement(s, k)
  count = empty map of letter to count   # [1] letters inside the window
  left = 0
  maxFreq = 0                            # [1] highest count seen. Never lowered
  for right from 0 to last index of s
    count[s[right]] = count[s[right]] + 1     # [2] take the new letter in
    maxFreq = max(maxFreq, count[s[right]])
    if (right - left + 1) - maxFreq > k       # [3] too many replacements
      count[s[left]] = count[s[left]] - 1
      left = left + 1                         # [3] slide the window right
  return length of s - left              # [4] window never shrinks: its size is the answer
