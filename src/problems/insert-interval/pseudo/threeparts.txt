function insert(intervals, newInterval)       # intervals: sorted, no overlaps
  result = empty list
  start, end = newInterval
  i = 0
  while i < n and intervals[i].end < start          # [1] ends before the new one starts
    append a copy of intervals[i] to result ; i = i + 1
  while i < n and intervals[i].start <= end         # [2] overlaps (touching counts)
    start = min(start, intervals[i].start)
    end   = max(end,   intervals[i].end)
    i = i + 1
  append [start, end] to result                     # [3] place the grown interval
  while i < n                                       # [4] the rest start after the new end
    append a copy of intervals[i] to result ; i = i + 1
  return result                                     # [4]
