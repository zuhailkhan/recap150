function search(nums, target)
  l = 0                         # [1] the live range is nums[l..r]
  r = length(nums) - 1          # [1]
  while l <= r                  # [2] something is still live
    m = l + (r - l) / 2         # [2] middle, rounded down
    if nums[m] == target        # [4] found it: that is the answer
      return m
    if nums[m] < target
      l = m + 1                 # [3] too small: drop the left half
    else
      r = m - 1                 # [3] too big: drop the right half
  return -1                     # [4] live range is empty: not found
