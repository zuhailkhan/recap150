function checkInclusion(s1, s2)
  n = length of s1
  need = empty map of letter to number     # [1] letters owed to the window
  off = 0                                  # [1] how many letters are not balanced
  for each letter c in s1
    change need[c] by +1 and update off    # [1] off changes when a count leaves or reaches 0
  for right from 0 to last index of s2
    if right >= n
      change need[s2[right - n]] by +1     # [2] old left letter leaves: owed again
    change need[s2[right]] by -1           # [3] new right letter enters: paid off
    if off == 0
      return true                          # [4] all balanced: a permutation
  return false
