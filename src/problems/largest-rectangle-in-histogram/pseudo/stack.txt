function largestRectangleArea(heights)
  stack = empty list                         # [1] indices of bars, heights rising upward
  best = 0
  for i from 0 to length(heights)            # one step past the end on purpose
    h = heights[i], or 0 when i is past the end   # [1] a height-0 bar pops everything
    while stack is not empty and h < heights[top of stack]
      top = pop stack                        # [2] a shorter bar ends the top bar's rectangle
      height = heights[top]
      left = top of stack, or -1 if empty    # [3] nearest shorter bar on the left
      width = i - left - 1                   # [3] bars strictly between left and i
      best = max(best, height * width)       # [3] keep the biggest area
    push i onto stack                        # [4] bar i can still grow to the right
  return best                                # [5] the biggest area seen
