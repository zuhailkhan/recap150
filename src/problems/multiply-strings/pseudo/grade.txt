function multiply(num1, num2)
  m = length of num1 ; n = length of num2
  res = array of m + n zeros                 # [1] a product has at most m + n digits
  for i from m - 1 down to 0                 # [2] each digit of num1, right to left
    for j from n - 1 down to 0
      p = digit(num1[i]) * digit(num2[j]) + res[i + j + 1]   # [3] digit product plus what the ones slot holds
      res[i + j + 1] = p mod 10              # [3] the ones digit stays in slot i + j + 1
      res[i + j] = res[i + j] + p div 10     # [3] the tens digit carries into slot i + j
  k = 0
  while k < m + n - 1 and res[k] is 0        # [4] skip leading zeros, but keep at least one digit
    k = k + 1
  return the digits res[k..] joined as text
