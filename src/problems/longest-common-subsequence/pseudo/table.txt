function lcs(text1, text2)
  m = length(text1), n = length(text2)
  dp = table of (m + 1) rows and (n + 1) columns, all 0    # [1] empty prefixes share nothing
  for i from 1 to m
    for j from 1 to n
      if text1[i-1] == text2[j-1]
        dp[i][j] = dp[i-1][j-1] + 1                        # [2] same letter: extend the diagonal
      else
        dp[i][j] = max(dp[i-1][j], dp[i][j-1])             # [3] different: best of above and left
  return dp[m][n]                                          # [4] both whole strings
