function threeSum(nums)
  a = sorted copy of nums                  # [1] equal values become neighbours
  result = empty list
  for i from 0 to length(a) - 3
    if i > 0 and a[i] == a[i - 1]          # [2] same first number as before
      continue                             # every triple from it would repeat
    l = i + 1, r = length(a) - 1           # [3] two pointers on the rest
    while l < r
      sum = a[i] + a[l] + a[r]
      if sum < 0
        l = l + 1                          # [3] too small: need a bigger number
      else if sum > 0
        r = r - 1                          # [3] too big: need a smaller number
      else
        add [a[i], a[l], a[r]] to result   # [4] record the triple
        l = l + 1, r = r - 1
        while l < r and a[l] == a[l - 1]
          l = l + 1                        # [5] skip repeats on the left
        while l < r and a[r] == a[r + 1]
          r = r - 1                        # [5] skip repeats on the right
  return result
