findKthLargest(nums, k)
  a = copy of nums
  target = len(a) - k                       # [1] index of the kth largest in sorted order
  lo, hi = 0, len(a) - 1
  loop
    p = partition(a, lo, hi)                # [2] pivot lands in its final sorted place
    if p == target: return a[p]             # [4] the pivot is in the target place
    if p < target: lo = p + 1               # [3] the target is on the right
    else: hi = p - 1                        # [3] the target is on the left

partition(a, lo, hi)
  pivot = a[middle]; move it to a[hi]
  i = lo
  for j in lo .. hi-1
    if a[j] < pivot: swap a[i], a[j]; i = i + 1
  swap a[i], a[hi]
  return i
