function lengthOfLIS(nums)
  n = length(nums)
  dp = list of n ones                    # [1] a lone number is a run of 1
  for i from 1 to n - 1
    for j from 0 to i - 1                # [2] every earlier number
      if nums[j] < nums[i]
        dp[i] = max(dp[i], dp[j] + 1)    # [3] extend the run that ends at j
  return the biggest value in dp         # [4] the run can end anywhere
