function isSubtree(root, subRoot)
  if subRoot is null                   # [1] an empty tree fits inside any tree
    return true
  if root is null                      # [1] ran out of tree without a match
    return false
  if sameTree(root, subRoot)           # [2] does the copy start exactly here?
    return true
  return isSubtree(root.left, subRoot) or isSubtree(root.right, subRoot)   # [3] try the children as new starting points

function sameTree(a, b)                # [2] shape and values must match all the way down
  if a is null and b is null
    return true
  if a is null or b is null
    return false
  return a.val == b.val and sameTree(a.left, b.left) and sameTree(a.right, b.right)
